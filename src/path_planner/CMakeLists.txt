cmake_minimum_required(VERSION 3.0.2)
project(path_planner)
find_package(catkin REQUIRED COMPONENTS nav_msgs geometry_msgs interfaces node_pipeline roscpp tf2_msgs lanelet2_core lanelet2_io lanelet2_routing lanelet2_traffic_rules)
find_package(pugixml REQUIRED CONFIG)
get_target_property(pugixml_INCLUDE_DIRS pugixml INTERFACE_INCLUDE_DIRECTORIES)
set(pugixml_LIBRARIES pugixml)
catkin_package(INCLUDE_DIRS include CATKIN_DEPENDS nav_msgs geometry_msgs interfaces node_pipeline roscpp tf2_msgs lanelet2_core lanelet2_io lanelet2_routing lanelet2_traffic_rules DEPENDS pugixml)
include_directories(include ${catkin_INCLUDE_DIRS} ${pugixml_INCLUDE_DIRS})
add_executable(path_planner_node src/ros_node.cpp)
target_compile_features(path_planner_node PRIVATE cxx_std_14)
target_compile_options(path_planner_node PRIVATE -Wall -Wextra -Wpedantic)
add_dependencies(path_planner_node ${catkin_EXPORTED_TARGETS})
target_link_libraries(path_planner_node ${catkin_LIBRARIES} ${pugixml_LIBRARIES})
if(CATKIN_ENABLE_TESTING)
  catkin_add_gtest(test_path_cores test/unit/test_path_cores.cpp)
  if(TARGET test_path_cores)
    target_compile_features(test_path_cores PRIVATE cxx_std_14)
    target_compile_definitions(test_path_cores PRIVATE "PATH_PLANNER_MAP_DIR=\"${CMAKE_CURRENT_SOURCE_DIR}/../map\"")
    add_dependencies(test_path_cores ${catkin_EXPORTED_TARGETS})
    target_link_libraries(test_path_cores ${catkin_LIBRARIES} ${pugixml_LIBRARIES})
  endif()
endif()
install(TARGETS path_planner_node RUNTIME DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION})
install(PROGRAMS launch.sh DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})
install(DIRECTORY include/${PROJECT_NAME}/ DESTINATION ${CATKIN_PACKAGE_INCLUDE_DESTINATION})
install(DIRECTORY config launch DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})
