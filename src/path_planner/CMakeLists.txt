cmake_minimum_required(VERSION 3.0.2)
project(path_planner)
find_package(catkin REQUIRED COMPONENTS geometry_msgs interfaces node_pipeline roscpp tf2_msgs)
catkin_package(INCLUDE_DIRS include CATKIN_DEPENDS geometry_msgs interfaces node_pipeline roscpp tf2_msgs)
include_directories(include ${catkin_INCLUDE_DIRS})
add_executable(path_planner_node src/ros_node.cpp)
target_compile_features(path_planner_node PRIVATE cxx_std_14)
target_compile_options(path_planner_node PRIVATE -Wall -Wextra -Wpedantic)
add_dependencies(path_planner_node ${catkin_EXPORTED_TARGETS})
target_link_libraries(path_planner_node ${catkin_LIBRARIES})
install(TARGETS path_planner_node RUNTIME DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION})
install(PROGRAMS launch.sh DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})
install(DIRECTORY include/${PROJECT_NAME}/ DESTINATION ${CATKIN_PACKAGE_INCLUDE_DESTINATION})
install(DIRECTORY config launch DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})
