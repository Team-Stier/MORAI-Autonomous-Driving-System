cmake_minimum_required(VERSION 3.0.2)
project(lidar_leveler)
find_package(catkin REQUIRED COMPONENTS geometry_msgs interfaces node_pipeline roscpp sensor_msgs tf2 tf2_msgs)
catkin_package(INCLUDE_DIRS include CATKIN_DEPENDS geometry_msgs interfaces node_pipeline roscpp sensor_msgs tf2 tf2_msgs)
include_directories(include ${catkin_INCLUDE_DIRS})
node_pipeline_add_node(lidar_leveler_node)
target_compile_features(lidar_leveler_node PRIVATE cxx_std_14)
target_compile_options(lidar_leveler_node PRIVATE -Wall -Wextra -Wpedantic)
add_dependencies(lidar_leveler_node ${catkin_EXPORTED_TARGETS})
target_link_libraries(lidar_leveler_node ${catkin_LIBRARIES})
if(CATKIN_ENABLE_TESTING)
  catkin_add_gtest(test_pass_through test/unit/test_pass_through.cpp)
  if(TARGET test_pass_through)
    target_compile_options(test_pass_through PRIVATE -include ${CMAKE_CURRENT_BINARY_DIR}/lidar_leveler_node_registration_numeric.hpp)
    add_dependencies(test_pass_through lidar_leveler_node_extension_contract)
    target_compile_features(test_pass_through PRIVATE cxx_std_14)
    add_dependencies(test_pass_through ${catkin_EXPORTED_TARGETS})
    target_link_libraries(test_pass_through ${catkin_LIBRARIES})
  endif()
  catkin_add_gtest(test_z_roi test/unit/test_z_roi.cpp)
  if(TARGET test_z_roi)
    target_compile_features(test_z_roi PRIVATE cxx_std_14)
    add_dependencies(test_z_roi ${catkin_EXPORTED_TARGETS})
    target_link_libraries(test_z_roi ${catkin_LIBRARIES} gtest_main)
  endif()
endif()
install(TARGETS lidar_leveler_node RUNTIME DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION})
install(PROGRAMS launch.sh DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})
install(DIRECTORY include/${PROJECT_NAME}/ DESTINATION ${CATKIN_PACKAGE_INCLUDE_DESTINATION})
install(DIRECTORY config launch DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})
