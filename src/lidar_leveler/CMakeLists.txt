cmake_minimum_required(VERSION 3.0.2)
project(lidar_leveler)
find_package(catkin REQUIRED COMPONENTS geometry_msgs interfaces node_pipeline roscpp sensor_msgs tf2_msgs)
catkin_package(INCLUDE_DIRS include CATKIN_DEPENDS geometry_msgs interfaces node_pipeline roscpp sensor_msgs tf2_msgs)
include_directories(include ${catkin_INCLUDE_DIRS})
add_executable(lidar_leveler_node src/ros_node.cpp)
target_compile_features(lidar_leveler_node PRIVATE cxx_std_14)
target_compile_options(lidar_leveler_node PRIVATE -Wall -Wextra -Wpedantic)
add_dependencies(lidar_leveler_node ${catkin_EXPORTED_TARGETS})
target_link_libraries(lidar_leveler_node ${catkin_LIBRARIES})
install(TARGETS lidar_leveler_node RUNTIME DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION})
install(PROGRAMS launch.sh DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})
install(DIRECTORY include/${PROJECT_NAME}/ DESTINATION ${CATKIN_PACKAGE_INCLUDE_DESTINATION})
install(DIRECTORY config launch DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})
