cmake_minimum_required(VERSION 3.0.2)
project(section_annotator)
find_package(catkin REQUIRED COMPONENTS nav_msgs geometry_msgs interfaces node_pipeline roscpp)
catkin_package(INCLUDE_DIRS include CATKIN_DEPENDS nav_msgs geometry_msgs interfaces node_pipeline roscpp)
include_directories(include ${catkin_INCLUDE_DIRS})
add_executable(section_annotator_node src/ros_node.cpp)
target_compile_features(section_annotator_node PRIVATE cxx_std_14)
target_compile_options(section_annotator_node PRIVATE -Wall -Wextra -Wpedantic)
add_dependencies(section_annotator_node ${catkin_EXPORTED_TARGETS})
target_link_libraries(section_annotator_node ${catkin_LIBRARIES})
if(CATKIN_ENABLE_TESTING)
  catkin_add_gtest(test_checkpoint_config test/unit/test_checkpoint_config.cpp)
  if(TARGET test_checkpoint_config)
    target_compile_features(test_checkpoint_config PRIVATE cxx_std_14)
    add_dependencies(test_checkpoint_config ${catkin_EXPORTED_TARGETS})
    target_link_libraries(test_checkpoint_config ${catkin_LIBRARIES})
  endif()
endif()
install(TARGETS section_annotator_node RUNTIME DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION})
install(PROGRAMS launch.sh DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})
install(DIRECTORY include/${PROJECT_NAME}/ DESTINATION ${CATKIN_PACKAGE_INCLUDE_DESTINATION})
install(DIRECTORY config launch DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})
