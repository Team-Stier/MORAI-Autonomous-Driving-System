cmake_minimum_required(VERSION 3.0.2)
project(traffic_light)
find_package(catkin REQUIRED COMPONENTS node_pipeline interfaces sensor_msgs)
find_package(OpenCV REQUIRED COMPONENTS core dnn imgcodecs imgproc)
set(_traffic_light_onnxruntime_root /opt/onnxruntime-gpu/1.19.2)
if(DEFINED ENV{ONNXRUNTIME_ROOT})
  set(_traffic_light_onnxruntime_root "$ENV{ONNXRUNTIME_ROOT}")
endif()
set(ONNXRUNTIME_ROOT "${_traffic_light_onnxruntime_root}" CACHE PATH "ONNX Runtime root")
find_path(ONNXRUNTIME_INCLUDE_DIR onnxruntime_cxx_api.h PATHS "${ONNXRUNTIME_ROOT}/include" NO_DEFAULT_PATH)
find_library(ONNXRUNTIME_LIBRARY onnxruntime PATHS "${ONNXRUNTIME_ROOT}/lib" NO_DEFAULT_PATH)
if(NOT ONNXRUNTIME_INCLUDE_DIR OR NOT ONNXRUNTIME_LIBRARY)
  message(FATAL_ERROR "ONNX Runtime not found under ONNXRUNTIME_ROOT=${ONNXRUNTIME_ROOT}")
endif()
set(CMAKE_BUILD_RPATH "${ONNXRUNTIME_ROOT}/lib;${ONNXRUNTIME_ROOT}/cuda/lib")
set(CMAKE_INSTALL_RPATH "${ONNXRUNTIME_ROOT}/lib;${ONNXRUNTIME_ROOT}/cuda/lib")
string(APPEND CMAKE_EXE_LINKER_FLAGS " -Wl,--disable-new-dtags")
string(APPEND CMAKE_SHARED_LINKER_FLAGS " -Wl,--disable-new-dtags")
np_add_nodes(INCLUDE_DIRS ${OpenCV_INCLUDE_DIRS} ${ONNXRUNTIME_INCLUDE_DIR}
             LIBRARIES ${OpenCV_LIBRARIES} ${ONNXRUNTIME_LIBRARY})
