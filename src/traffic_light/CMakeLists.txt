cmake_minimum_required(VERSION 3.0.2)
project(traffic_light)
find_package(catkin REQUIRED COMPONENTS geometry_msgs interfaces node_pipeline roscpp sensor_msgs)
find_package(OpenCV REQUIRED COMPONENTS core dnn imgcodecs imgproc)
find_package(yaml-cpp REQUIRED)
set(ONNXRUNTIME_ROOT /opt/onnxruntime-gpu/1.19.2 CACHE PATH "ONNX Runtime root")
find_path(ONNXRUNTIME_INCLUDE_DIR onnxruntime_cxx_api.h PATHS ${ONNXRUNTIME_ROOT}/include NO_DEFAULT_PATH REQUIRED)
find_library(ONNXRUNTIME_LIBRARY onnxruntime PATHS ${ONNXRUNTIME_ROOT}/lib NO_DEFAULT_PATH REQUIRED)
catkin_package(INCLUDE_DIRS include CATKIN_DEPENDS geometry_msgs interfaces node_pipeline roscpp sensor_msgs)
include_directories(include ${catkin_INCLUDE_DIRS} ${OpenCV_INCLUDE_DIRS})
include_directories(SYSTEM ${ONNXRUNTIME_INCLUDE_DIR})
add_executable(traffic_light_node src/ros_node.cpp src/detection.cpp src/yolo_traffic_light_core.cpp)
target_compile_features(traffic_light_node PRIVATE cxx_std_14)
target_compile_options(traffic_light_node PRIVATE -Wall -Wextra -Wpedantic)
add_dependencies(traffic_light_node ${catkin_EXPORTED_TARGETS})
target_link_libraries(traffic_light_node ${catkin_LIBRARIES} ${OpenCV_LIBRARIES} ${ONNXRUNTIME_LIBRARY} yaml-cpp)
set_target_properties(traffic_light_node PROPERTIES BUILD_RPATH
  "${ONNXRUNTIME_ROOT}/lib;${ONNXRUNTIME_ROOT}/cuda/lib" LINK_FLAGS "-Wl,--disable-new-dtags")
if(CATKIN_ENABLE_TESTING)
  catkin_add_gtest(test_traffic_light_detection test/unit/test_detection.cpp src/detection.cpp)
  if(TARGET test_traffic_light_detection)
    target_compile_features(test_traffic_light_detection PRIVATE cxx_std_14)
    target_compile_options(test_traffic_light_detection PRIVATE -Wall -Wextra -Wpedantic)
    target_link_libraries(test_traffic_light_detection ${catkin_LIBRARIES} ${OpenCV_LIBRARIES})
  endif()
  catkin_add_gtest(test_yolo_traffic_light_core test/unit/test_yolo_traffic_light_core.cpp
    src/yolo_traffic_light_core.cpp src/detection.cpp)
  if(TARGET test_yolo_traffic_light_core)
    target_compile_features(test_yolo_traffic_light_core PRIVATE cxx_std_14)
    target_compile_options(test_yolo_traffic_light_core PRIVATE -Wall -Wextra -Wpedantic)
    target_compile_definitions(test_yolo_traffic_light_core PRIVATE
      "TRAFFIC_LIGHT_FIXTURE_DIR=\"${CMAKE_CURRENT_SOURCE_DIR}/test/fixtures\"")
    target_link_libraries(test_yolo_traffic_light_core ${catkin_LIBRARIES} ${OpenCV_LIBRARIES} ${ONNXRUNTIME_LIBRARY} yaml-cpp)
    set_target_properties(test_yolo_traffic_light_core PROPERTIES BUILD_RPATH
      "${ONNXRUNTIME_ROOT}/lib;${ONNXRUNTIME_ROOT}/cuda/lib" LINK_FLAGS "-Wl,--disable-new-dtags")
  endif()
  catkin_add_gtest(test_traffic_light_config test/unit/test_config.cpp)
  if(TARGET test_traffic_light_config)
    target_compile_features(test_traffic_light_config PRIVATE cxx_std_14)
    target_compile_options(test_traffic_light_config PRIVATE -Wall -Wextra -Wpedantic)
    target_link_libraries(test_traffic_light_config ${catkin_LIBRARIES})
  endif()
  catkin_add_gtest(test_yolo_core_registration test/unit/test_yolo_core_registration.cpp
    src/yolo_traffic_light_core.cpp src/detection.cpp)
  if(TARGET test_yolo_core_registration)
    target_compile_features(test_yolo_core_registration PRIVATE cxx_std_14)
    target_compile_options(test_yolo_core_registration PRIVATE -Wall -Wextra -Wpedantic)
    target_compile_definitions(test_yolo_core_registration PRIVATE
      "TRAFFIC_LIGHT_FIXTURE_DIR=\"${CMAKE_CURRENT_SOURCE_DIR}/test/fixtures\"")
    target_link_libraries(test_yolo_core_registration ${catkin_LIBRARIES} ${OpenCV_LIBRARIES} ${ONNXRUNTIME_LIBRARY} yaml-cpp)
    set_target_properties(test_yolo_core_registration PROPERTIES BUILD_RPATH
      "${ONNXRUNTIME_ROOT}/lib;${ONNXRUNTIME_ROOT}/cuda/lib" LINK_FLAGS "-Wl,--disable-new-dtags")
  endif()
endif()
install(TARGETS traffic_light_node RUNTIME DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION})
install(PROGRAMS launch.sh DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})
install(DIRECTORY include/${PROJECT_NAME}/ DESTINATION ${CATKIN_PACKAGE_INCLUDE_DESTINATION})
install(DIRECTORY config launch models DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})
