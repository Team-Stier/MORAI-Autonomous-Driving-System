cmake_minimum_required(VERSION 3.0.2)
project(visualizer)
find_package(catkin REQUIRED COMPONENTS interfaces rospy geometry_msgs sensor_msgs std_msgs visualization_msgs)
catkin_package(CATKIN_DEPENDS interfaces rospy geometry_msgs sensor_msgs std_msgs visualization_msgs)
catkin_install_python(PROGRAMS scripts/objects_rviz.py DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION})
if(CATKIN_ENABLE_TESTING)
  catkin_add_nosetests(test/unit/test_objects_rviz.py)
endif()
install(PROGRAMS launch.sh DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})
install(DIRECTORY launch config DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})
