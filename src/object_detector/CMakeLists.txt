cmake_minimum_required(VERSION 3.0.2)
project(object_detector)
find_package(catkin REQUIRED COMPONENTS geometry_msgs interfaces node_pipeline roscpp sensor_msgs)
find_package(PCL 1.10 REQUIRED COMPONENTS common filters segmentation features registration search)
catkin_package(INCLUDE_DIRS include CATKIN_DEPENDS geometry_msgs interfaces node_pipeline roscpp sensor_msgs DEPENDS PCL)
include_directories(include ${catkin_INCLUDE_DIRS} ${PCL_INCLUDE_DIRS})
add_definitions(${PCL_DEFINITIONS})
node_pipeline_add_node(object_detector_node)
target_compile_features(object_detector_node PRIVATE cxx_std_14)
target_compile_options(object_detector_node PRIVATE -Wall -Wextra -Wpedantic)
add_dependencies(object_detector_node ${catkin_EXPORTED_TARGETS})
target_link_libraries(object_detector_node ${catkin_LIBRARIES} ${PCL_LIBRARIES})
if(CATKIN_ENABLE_TESTING)
  catkin_add_gtest(test_two_core_pipeline test/unit/test_two_core_pipeline.cpp)
  if(TARGET test_two_core_pipeline)
    target_compile_options(test_two_core_pipeline PRIVATE -include ${CMAKE_CURRENT_BINARY_DIR}/object_detector_node_registration_numeric.hpp)
    add_dependencies(test_two_core_pipeline object_detector_node_extension_contract)
    target_compile_features(test_two_core_pipeline PRIVATE cxx_std_14)
    add_dependencies(test_two_core_pipeline ${catkin_EXPORTED_TARGETS})
    target_link_libraries(test_two_core_pipeline ${catkin_LIBRARIES} ${PCL_LIBRARIES} gtest_main)
  endif()
  catkin_add_gtest(test_cluster_match test/unit/test_cluster_match.cpp)
  if(TARGET test_cluster_match)
    target_compile_features(test_cluster_match PRIVATE cxx_std_14)
    add_dependencies(test_cluster_match ${catkin_EXPORTED_TARGETS})
    target_link_libraries(test_cluster_match ${catkin_LIBRARIES} ${PCL_LIBRARIES} gtest_main)
  endif()
  catkin_add_gtest(test_tracker test/unit/test_tracker.cpp)
  if(TARGET test_tracker)
    target_compile_features(test_tracker PRIVATE cxx_std_14)
    add_dependencies(test_tracker ${catkin_EXPORTED_TARGETS})
    target_link_libraries(test_tracker ${catkin_LIBRARIES} ${PCL_LIBRARIES} gtest_main)
  endif()
  catkin_add_gtest(test_npc_model_bank test/unit/test_npc_model_bank.cpp)
  if(TARGET test_npc_model_bank)
    target_compile_features(test_npc_model_bank PRIVATE cxx_std_14)
    add_dependencies(test_npc_model_bank ${catkin_EXPORTED_TARGETS})
    target_link_libraries(test_npc_model_bank ${catkin_LIBRARIES} ${PCL_LIBRARIES} gtest_main)
  endif()
endif()
install(TARGETS object_detector_node RUNTIME DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION})
install(PROGRAMS launch.sh DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})
install(DIRECTORY include/${PROJECT_NAME}/ DESTINATION ${CATKIN_PACKAGE_INCLUDE_DESTINATION})
install(DIRECTORY config launch DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})
