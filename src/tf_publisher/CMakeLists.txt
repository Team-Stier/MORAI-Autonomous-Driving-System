cmake_minimum_required(VERSION 3.0.2)
project(tf_publisher)
find_package(catkin REQUIRED COMPONENTS nav_msgs geometry_msgs interfaces node_pipeline roscpp tf2_msgs)
catkin_package(INCLUDE_DIRS include CATKIN_DEPENDS nav_msgs geometry_msgs interfaces node_pipeline roscpp tf2_msgs)
include_directories(include ${catkin_INCLUDE_DIRS})
node_pipeline_add_node(tf_publisher_node)
node_pipeline_add_node(tf_static_publisher_node NODE static_node.yaml)
foreach(target tf_publisher_node tf_static_publisher_node)
  target_compile_features(${target} PRIVATE cxx_std_14)
  target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
  add_dependencies(${target} ${catkin_EXPORTED_TARGETS})
  target_link_libraries(${target} ${catkin_LIBRARIES})
endforeach()
if(CATKIN_ENABLE_TESTING)
  add_executable(test_static_tf test/unit/test_static_tf.cpp)
  target_compile_features(test_static_tf PRIVATE cxx_std_14)
  target_compile_options(test_static_tf PRIVATE -UNDEBUG)
  target_link_libraries(test_static_tf ${catkin_LIBRARIES})
  add_test(NAME test_static_tf COMMAND test_static_tf)
endif()
install(TARGETS tf_publisher_node tf_static_publisher_node RUNTIME DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION})
install(PROGRAMS launch.sh DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})
install(DIRECTORY include/${PROJECT_NAME}/ DESTINATION ${CATKIN_PACKAGE_INCLUDE_DESTINATION})
install(DIRECTORY config launch DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})
