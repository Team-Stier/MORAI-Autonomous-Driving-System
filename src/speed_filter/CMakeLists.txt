cmake_minimum_required(VERSION 3.0.2)
project(speed_filter)
find_package(catkin REQUIRED COMPONENTS geometry_msgs interfaces morai_msgs node_pipeline roscpp)
catkin_package(INCLUDE_DIRS include CATKIN_DEPENDS geometry_msgs interfaces morai_msgs node_pipeline roscpp)
include_directories(include ${catkin_INCLUDE_DIRS})
node_pipeline_add_node(speed_filter_node)
target_compile_features(speed_filter_node PRIVATE cxx_std_14)
target_compile_options(speed_filter_node PRIVATE -Wall -Wextra -Wpedantic)
add_dependencies(speed_filter_node ${catkin_EXPORTED_TARGETS})
target_link_libraries(speed_filter_node ${catkin_LIBRARIES})
if(CATKIN_ENABLE_TESTING)
  catkin_add_gtest(test_speed_filter test/unit/test_speed_filter.cpp)
  if(TARGET test_speed_filter)
    target_compile_options(test_speed_filter PRIVATE -include ${CMAKE_CURRENT_BINARY_DIR}/speed_filter_node_registration_numeric.hpp)
    add_dependencies(test_speed_filter speed_filter_node_extension_contract)
    target_compile_features(test_speed_filter PRIVATE cxx_std_14)
    add_dependencies(test_speed_filter ${catkin_EXPORTED_TARGETS})
    target_link_libraries(test_speed_filter ${catkin_LIBRARIES})
  endif()
endif()
install(TARGETS speed_filter_node RUNTIME DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION})
install(PROGRAMS launch.sh DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})
install(DIRECTORY include/${PROJECT_NAME}/ DESTINATION ${CATKIN_PACKAGE_INCLUDE_DESTINATION})
install(DIRECTORY config launch DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})
