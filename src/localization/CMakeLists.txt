cmake_minimum_required(VERSION 3.0.2)
project(localization)
find_package(catkin REQUIRED COMPONENTS node_pipeline roscpp sensor_msgs geometry_msgs interfaces)
find_package(Eigen3 REQUIRED)
catkin_package(INCLUDE_DIRS include CATKIN_DEPENDS node_pipeline roscpp sensor_msgs geometry_msgs interfaces DEPENDS Eigen3)
include_directories(include ${catkin_INCLUDE_DIRS} ${EIGEN3_INCLUDE_DIRS})
add_executable(localization_node src/localization_node.cpp)
target_compile_features(localization_node PRIVATE cxx_std_14)
target_compile_options(localization_node PRIVATE -Wall -Wextra -Wpedantic)
add_dependencies(localization_node ${catkin_EXPORTED_TARGETS})
target_link_libraries(localization_node ${catkin_LIBRARIES})
install(TARGETS localization_node RUNTIME DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION})
install(PROGRAMS launch.sh DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})
install(DIRECTORY include/${PROJECT_NAME}/ DESTINATION ${CATKIN_PACKAGE_INCLUDE_DESTINATION})
install(DIRECTORY config launch DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})
