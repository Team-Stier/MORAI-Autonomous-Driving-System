cmake_minimum_required(VERSION 3.0.2)
project(localization)
find_package(catkin REQUIRED COMPONENTS nav_msgs node_pipeline roscpp sensor_msgs geometry_msgs interfaces)
find_package(Eigen3 REQUIRED)
catkin_package(INCLUDE_DIRS include CATKIN_DEPENDS nav_msgs node_pipeline roscpp sensor_msgs geometry_msgs interfaces DEPENDS Eigen3)
include_directories(include ${catkin_INCLUDE_DIRS} ${EIGEN3_INCLUDE_DIRS})
node_pipeline_add_node(localization_node)
target_compile_features(localization_node PRIVATE cxx_std_14)
target_compile_options(localization_node PRIVATE -Wall -Wextra -Wpedantic)
add_dependencies(localization_node ${catkin_EXPORTED_TARGETS})
target_link_libraries(localization_node ${catkin_LIBRARIES})
install(TARGETS localization_node RUNTIME DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION})
install(PROGRAMS launch.sh DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})
install(DIRECTORY include/${PROJECT_NAME}/ DESTINATION ${CATKIN_PACKAGE_INCLUDE_DESTINATION})
install(DIRECTORY config launch DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})
if(CATKIN_ENABLE_TESTING)
  catkin_add_gtest(test_full_ekf test/unit/test_full_ekf.cpp)
  if(TARGET test_full_ekf)
    target_compile_features(test_full_ekf PRIVATE cxx_std_14)
    add_dependencies(test_full_ekf ${catkin_EXPORTED_TARGETS})
    target_link_libraries(test_full_ekf ${catkin_LIBRARIES})
  endif()
  catkin_add_gtest(test_tunnel_wall test/unit/test_tunnel_wall.cpp)
  if(TARGET test_tunnel_wall)
    target_compile_options(test_tunnel_wall PRIVATE -include ${CMAKE_CURRENT_BINARY_DIR}/localization_node_registration_numeric.hpp)
    add_dependencies(test_tunnel_wall localization_node_extension_contract)
    target_compile_features(test_tunnel_wall PRIVATE cxx_std_14)
    target_include_directories(test_tunnel_wall PRIVATE ${catkin_INCLUDE_DIRS})
    add_dependencies(test_tunnel_wall ${catkin_EXPORTED_TARGETS})
    target_link_libraries(test_tunnel_wall ${catkin_LIBRARIES} gtest_main)
  endif()
endif()
