cmake_minimum_required(VERSION 3.0.2)
project(velocity_planner)
find_package(catkin REQUIRED COMPONENTS nav_msgs geometry_msgs interfaces node_pipeline roscpp tf2 tf2_geometry_msgs tf2_msgs)
catkin_package(INCLUDE_DIRS include CATKIN_DEPENDS nav_msgs geometry_msgs interfaces node_pipeline roscpp tf2 tf2_geometry_msgs tf2_msgs)
include_directories(include ${catkin_INCLUDE_DIRS})
node_pipeline_add_node(velocity_planner_node)
target_compile_features(velocity_planner_node PRIVATE cxx_std_14)
target_compile_options(velocity_planner_node PRIVATE -Wall -Wextra -Wpedantic)
add_dependencies(velocity_planner_node ${catkin_EXPORTED_TARGETS})
target_link_libraries(velocity_planner_node ${catkin_LIBRARIES})
if(CATKIN_ENABLE_TESTING)
  catkin_add_gtest(test_plain_velocity test/unit/test_plain_velocity.cpp)
  if(TARGET test_plain_velocity)
    target_compile_options(test_plain_velocity PRIVATE -include ${CMAKE_CURRENT_BINARY_DIR}/velocity_planner_node_registration_numeric.hpp)
    add_dependencies(test_plain_velocity velocity_planner_node_extension_contract)
    target_compile_features(test_plain_velocity PRIVATE cxx_std_14)
    add_dependencies(test_plain_velocity ${catkin_EXPORTED_TARGETS})
    target_link_libraries(test_plain_velocity ${catkin_LIBRARIES})
  endif()
endif()
install(TARGETS velocity_planner_node RUNTIME DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION})
install(PROGRAMS launch.sh DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})
install(DIRECTORY include/${PROJECT_NAME}/ DESTINATION ${CATKIN_PACKAGE_INCLUDE_DESTINATION})
install(DIRECTORY config launch DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})
